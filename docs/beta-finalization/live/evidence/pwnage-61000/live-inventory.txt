# Nuxel S10+ (beyond2lte): disable WirePlumber's V4L2/libcamera monitor.
#
# The downstream Samsung MFC exposes /dev/video* nodes whose open() calls
# request_firmware() and blocks in mfc_load_firmware() because the vendor
# firmware is absent.  WirePlumber's video-capture monitor scans those nodes at
# session start and enters uninterruptible D state, which makes wpctl and the
# audio policy hang.  Nuxel has no working camera path yet, so disable only the
# video-capture feature; audio and Bluetooth remain required.
#
# The full main profile is repeated deliberately: a profile table override must
# not silently drop hardware.audio or hardware.bluetooth when it disables one
# feature.
wireplumber.profiles = {
  main = {
    inherits = [ base ]

    metadata.sm-settings = required
    metadata.sm-objects = required

    policy.standard = required

    hardware.audio = required
    hardware.bluetooth = required
    hardware.video-capture = disabled
  }
}
