#!/usr/bin/env python3
"""nuxel-orientationd — the ONE authoritative orientation policy service (A49).

Persistence, mode validation, sensor bring-up and compositor apply/revert live
here so Quick Settings, Settings, nuxelctl and the compositor share one source of
truth. Manual/fixed modes work with no sensor; auto modes report
``sensor_unavailable`` when no verified source exists. Nothing flashes, writes
partitions, or reboots.

CLI:
    nuxel-orientationd status [--json]      mode + effective + sensor + compositor
    nuxel-orientationd get                  selected mode
    nuxel-orientationd modes                the seven named modes + allowed sets
    nuxel-orientationd set <mode> [--apply] validate, persist, attempt transform
    nuxel-orientationd selftest             offline policy/sensor sanity
    nuxel-orientationd watch [seconds]      bounded auto-rotation loop (hardware)

Paths resolve through NUXEL_ROOT (default /) so the offline harness runs the REAL
code against a fake root. The exact mount matrix / transform mapping stay
``unverified`` until an owner physically proves them.
"""

def load_config():
    path = P(CONFIG_PATH)
    try:
        with open(path) as f:
            doc = json.load(f)
    except (OSError, ValueError):
        doc = {}
    merged = json.loads(json.dumps(DEFAULT_CONFIG))
    if isinstance(doc, dict):
        for key in ("modes", "policy", "sensor", "compositor", "paths",
                    "mode_order", "default_mode"):
            if key in doc:
                merged[key] = doc[key]
    merged["modes"] = {**SEVEN_MODES, **merged.get("modes", {})}
    merged["mode_order"] = MODE_ORDER
    return merged


def state_path(config):
    return P(config.get("paths", {}).get("state_file",
           "/var/lib/nuxel/orientation/state.json"))


def load_state(config):
    path = state_path(config)
    try:
        with open(path) as f:
            doc = json.load(f)
    except (OSError, ValueError):
        doc = {}
    if not isinstance(doc, dict):
        doc = {}
    return doc


def save_state(config, doc):
    path = state_path(config)
    os.makedirs(os.path.dirname(path), exist_ok=True)
    tmp = path + ".tmp"
    with open(tmp, "w") as f:
        json.dump(doc, f, indent=2)
        f.write("\n")
    os.replace(tmp, path)


def sensor_probe(config):
    """Per-source availability, honestly, without fabricating a reading.

    ``available`` reflects ONLY auto-capable sources (a real gravity source),
    never the diagnostic `manual` source, so an auto mode is reported
    ``sensor_unavailable`` unless a verified live accelerometer exists.
    """
    report = sensor_mod.probe_sources(config)
    capable = [s for s in report if s.get("auto_capable")]
    return {
        "sources": report,
        "available": bool(capable),
        "preferred": capable[0]["source"] if capable else None,
        "unverified": (config.get("sensor", {}).get("mount_matrix", {})
                       .get("calibration_status", "unverified") == "unverified"),
    }


def sensor_reading(config, source_name):
    src = sensor_mod.resolve_source(source_name, config)
    return src.read()


def do_status(config):

    sensor = sensor_probe(config)

        "auto": is_auto(mode),

    return result


def do_set(config, mode, apply_now=False):
    mode = validate_mode(mode)
    state = load_state(config)
    _, _, last = normalize_mode(state)
    sensor_orient = None
    if apply_now and is_auto(mode):
        # Choose a sensor-consistent initial pose when we can, never fabricate.
        probe = sensor_probe(config)
        pref = probe.get("preferred")
        if pref:
            try:
                reading = sensor_reading(config, pref)
                if reading.available:
                    from nuxel_orientation.policy import Orientator
                    o = Orientator(mode, config)
                    sensor_orient, _ = o.classify(reading.vector())
            except Exception:
                sensor_orient = None
    initial = choose_initial_orientation(mode, last, sensor_orient)
    next_state = {
        "schema_version": 1,
        "mode": mode,
        "effective_orientation": initial,
        "last_orientation": initial,
        "set_by": "nuxel-orientationd",
        "updated_at": int(time.time()),
    }
    save_state(config, next_state)

    apply_result = None
    if apply_now:
        apply_result = apply_mod.apply_transform(initial, config)
    result = {"success": True, "mode": mode, "effective_orientation": initial,
              "auto": is_auto(mode)}
    if apply_result is not None:
        result["compositor"] = apply_result
    return result


def do_modes():
    return [
        {"mode": name, "label": SEVEN_MODES[name]["label"],
         "allowed": list(SEVEN_MODES[name]["allowed"]),
         "auto": len(SEVEN_MODES[name]["allowed"]) > 1}
        for name in MODE_ORDER
    ]


def do_watch(config, seconds):
    probe = sensor_probe(config)
    if not probe.get("available"):
        print("sensor_unavailable: " + json.dumps(probe))
        return 2
    pref = probe["preferred"]
    state = load_state(config)
    mode, _, last = normalize_mode(state)
    if not is_auto(mode):
        print("mode %s is fixed; nothing to auto-rotate" % mode)
        return 0
    orientator = Orientator(mode, config)
    interval = float(config.get("policy", {}).get("sample_interval_ms", 100)) / 1000.0
    deadline = time.monotonic() + float(seconds)
    n = 0
    while time.monotonic() < deadline:
        try:
            reading = sensor_reading(config, pref)
        except Exception as exc:
            print("sensor read failed: %s" % exc)
            return 2
        changed = orientator.update(reading.vector())
        n += 1
        if changed is not None:
            print("orientation -> %d (samples=%d)" % (changed, n))
            save_state(config, {"mode": mode, "effective_orientation": changed,
                                "last_orientation": changed,
                                "updated_at": int(time.time())})
        time.sleep(interval)
    print("watch done: %d samples, effective=%d"
          % (n, orientator.status()["effective_orientation"]))
    return 0
