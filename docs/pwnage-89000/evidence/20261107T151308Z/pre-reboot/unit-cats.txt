# /etc/systemd/system/nuxel-boot-sound.service
[Unit]
Description=Nuxel startup cue after confirmed boot-animation to launcher handoff
After=nuxel-wayland.service nuxel-wl-fb.service sound.target nuxel-button-manager.service
Wants=nuxel-wayland.service nuxel-wl-fb.service
ConditionPathExists=/usr/share/nuxel/assets/nuxel-pow.mp3
# PWNAGE-89000: wait for the applied-route receipt, not the pending state
# document.  This service is event-triggered by nuxel-boot-sound.path, so it is
# no longer a 30+ s oneshot inside the initial boot transaction.  The played
# marker prevents a later route re-apply from replaying the boot cue.
ConditionPathExists=/run/nuxel-audio-route.ready
ConditionPathExists=!/run/nuxel-boot-sound.played
# The sound-cue binary records audibility as unverified; no physical claim is
# made here.

[Service]
Type=oneshot
# --wait polls the SAME read-only checks (calliope MUB0, both DSP1s,
# every route control, 48 kHz sink, default sink, volume cap) for up to
# 200 s.  A one-shot --check here failed at 09:35:45Z, 0.1 s after the
# ABOX reload, before the CS35L41 DSP1 download had finished.
# Bounded: a cold ABOX DSP download must never hold the boot transaction
# open forever. The cue is cosmetic; the ABOX timer can finish later.
ExecStartPre=-/usr/lib/nuxel/nuxel-audio-route.sh --check
ExecStart=/usr/libexec/nuxel/nuxel-sound-cue boot
ExecStartPost=/usr/bin/touch /run/nuxel-boot-sound.played
# The cue process itself repeats the read-only `nuxel-audio-route.sh --check`
# readiness gate immediately before playback; a stale state file is
# deliberately NOT sufficient.
# Playback is diagnostic until the physical route is verified. It must never
# hold the boot target hostage; nuxel-sound-cue itself bounds a player attempt
# to 11 seconds and reaps its process group.
# The cue waits (bounded, 180 s inside the program) for the live route check
# first.  A cold boot needs the ABOX PM reload plus both CS35L41 DSP1 firms to
# boot; the old 45 s service timeout killed the cue before that could finish,
# which is exactly what failed on 2026-10-30 09:35:45Z.  pw-play is separately
# bounded inside nuxel-sound-cue.
TimeoutStartSec=240
TimeoutStopSec=10
StandardOutput=journal
StandardError=journal
# Deliberately NOT Restart=always: play once, not on every bridge crash.

[Install]
# Enabled through nuxel-boot-sound.path (paths.target), not through the Wayland
# session: the cue must not hold the compositor/launcher transaction open.

[Service]
# nuxel-audio-warmup holds an inaudible stream briefly so the 366 ms cue is

# /etc/systemd/system/nuxel-boot-sound.service.d/30-defer-abox.conf
# PWNAGE-51000: the boot cue is cosmetic.  It must not pull the 20-50 s ABOX
# firmware reload or the audio-route oneshot into the initial boot transaction.
# The ABOX timer runs after Home and starts the route when the DSP is ready.
[Unit]
After=nuxel-wayland.service nuxel-wl-fb.service sound.target nuxel-button-manager.service
Wants=nuxel-wayland.service nuxel-wl-fb.service

[Service]
# Fast, non-fatal readiness check; the cue itself rechecks before playback.
ExecStartPre=-/usr/lib/nuxel/nuxel-audio-route.sh --check

# /etc/systemd/system/nuxel-boot-sound.service.d/40-after-pipewire.conf
# The cue used to select its player at import time, before PipeWire had created
# /run/nuxel-wayland/pipewire-0, and then fell back to direct ALSA (EINVAL).
# Start it only after the standalone audio session, and treat this cosmetic cue
# as non-fatal so a cold ABOX/DSP route cannot leave a failed unit at boot.
[Unit]
After=nuxel-audio-session.service
Wants=nuxel-audio-session.service

[Service]
ExecStartPre=/usr/bin/sh -c 'for i in $(seq 1 50); do [ -S /run/nuxel-wayland/pipewire-0 ] && exit 0; sleep 0.2; done; exit 0'
ExecStart=
ExecStart=-/usr/libexec/nuxel/nuxel-sound-cue boot

# /etc/systemd/system/nuxel-boot-sound.service.d/99-p80000-fast-stop.conf
[Service]
# The boot cue is cosmetic and may still be waiting on the ABOX/DSP route when
# the owner reboots.  It must never hold the shutdown transaction for 20 s.
TimeoutStopSec=5s
# Keep the oneshot lifecycle attached to the cue process only.  20-warmup-
# keepalive.conf intentionally leaves a short-lived nuxel-audio-warmup helper
# in the cgroup; control-group teardown waits for it and previously marked a
# successful cue as failed after this 5 s timeout.
KillMode=process
SendSIGKILL=yes

# /etc/systemd/system/nuxel-boot-sound.path
[Unit]
Description=Play the Nuxel boot cue once the audio route is actually ready
Documentation=file:/usr/share/doc/nuxel/SOUNDS.md

[Path]
# nuxel-audio-route.sh creates this receipt only after the ABOX route and both
# CS35L41 DSPs verify.  Use PathChanged, not PathExists: the receipt appears
# once per boot and a persistent PathExists trigger would re-trigger a skipped
# service until systemd's trigger limit.  The service marker is a second guard.
PathChanged=/run/nuxel-audio-route.ready
Unit=nuxel-boot-sound.service

[Install]
WantedBy=paths.target

# /etc/systemd/system/nuxel-web.service
[Unit]
Description=Nuxel PWA Static Web Server (nuxel-home launcher on :8000)
Documentation=https://nuxel.org
# PWNAGE-32000 (boot regression): the static server is loopback-only and must
# not wait for network.target/NetworkManager to reach the launcher.
# PWNAGE-89000: the PWA document itself is static.  The bridge/API are pulled in
# concurrently and the launcher retries bridge calls after the document is
# painted.  Ordering web behind nuxel-http-bridge added ~3 s to first paint.
Wants=nuxel-http-bridge.service
# The docroot is resolved by the shim: /usr/share/nuxel on the phone rootfs,
# /home/nuxel/nuxel on the flash-free deploy mirror (or $NUXEL_WEBROOT).
ConditionPathExists=/usr/bin/nuxel-web
# Restart-burst bound.  These keys belong to [Unit]; in [Service] systemd
# ignores them (measured journal: "Unknown key 'StartLimitIntervalSec'").
StartLimitIntervalSec=60
StartLimitBurst=5

[Service]
Type=simple
# /etc/nuxel/web.conf is written by the rootfs builder (NUXEL_WEBROOT,
# NUXEL_WEB_PORT, NUXEL_WEB_BIND). It is loaded here so the documented docroot
# override is real rather than inert; the leading '-' keeps the unit valid when
# the file is absent (VM / development hosts).
EnvironmentFile=-/etc/nuxel/web.conf
# Serve the FULL Nuxel PWA tree from the repo mirror the deploy staged at
# /home/nuxel/nuxel. Serving from the repo ROOT (not pwa/nuxel-home) is
# REQUIRED so the shared bridge client pwa/nuxel-common/bridge.js resolves via
# the PWA's relative `../nuxel-common/bridge.js` reference (the design enforced
# by pwa/nuxel-common/verify-wiring.js). The home launcher is therefore served
# at /pwa/nuxel-home/index.html. Runs as the nuxel user so the PWA tree is
# readable.
User=nuxel
Group=nuxel
# The S10+ 4.14 Android-derived kernel enables the legacy AID_INET gate:
# unprivileged processes need numeric gid 3003 to create AF_INET sockets.
# Keep the static server unprivileged and grant only that socket capability.
SupplementaryGroups=inet
Environment=NUXEL_WEB_PORT=8000
Environment=NUXEL_WEB_BIND=127.0.0.1
ExecStart=/usr/bin/nuxel-web
Restart=on-failure
RestartSec=2

# Hardening (localhost-only; PWA static content is read-only)
ProtectSystem=strict
ProtectHome=read-only
PrivateTmp=yes
PrivateDevices=yes
NoNewPrivileges=yes
RestrictAddressFamilies=AF_INET AF_INET6
ReadOnlyPaths=/
ReadOnlyPaths=/usr/share/nuxel
# The handset serves /usr/share/nuxel and does not carry the VM/deploy mirror.
# A missing optional mirror must not fail mount-namespace setup (226/NAMESPACE).
ReadWritePaths=-/home/nuxel/nuxel

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/nuxel-soundd.service
[Unit]
Description=Nuxel central sound manager (persistent PipeWire cue stream)
Documentation=file:/usr/share/doc/nuxel/SOUNDS.md
After=local-fs.target
Wants=nuxel-audio-session.service
StartLimitIntervalSec=60
StartLimitBurst=8

[Service]
# PWNAGE-89000: the control socket is the readiness contract.  The daemon binds
# and listens before it talks to PipeWire, so alarms can be accepted/queued while
# the S10+ ABOX/CS35L41 hardware route is still initializing.  The worker reports
# backend state separately through the socket; it never fakes hardware readiness.
Type=notify
NotifyAccess=main
Environment=XDG_RUNTIME_DIR=/run/nuxel-wayland
Environment=PIPEWIRE_RUNTIME_DIR=/run/nuxel-wayland
Environment=DBUS_SESSION_BUS_ADDRESS=unix:path=/run/nuxel-wayland/bus
Environment=PULSE_SERVER=unix:/run/nuxel-wayland/pulse/native
# PWNAGE-89000 ROOT CAUSE: this used to be
#   ExecStartPre=-/usr/lib/nuxel/nuxel-audio-route.sh --wait 200
# which held the whole service (and multi-user.target) for 40+ s while the ABOX
# core firmware and both CS35L41 DSP1s booted.  Route readiness is now handled
# asynchronously by nuxel-audio-route.service/timer and the sound worker.
# Idempotent: installs newly supplied files, preserves owner-edited files and
# creates the assignments document only when it is absent.
ExecStartPre=/usr/libexec/nuxel/install-sound-library
ExecStart=/usr/libexec/nuxel/nuxel-soundd
Restart=on-failure
RestartSec=2
TimeoutStopSec=5
TimeoutStartSec=30
KillMode=control-group
SendSIGKILL=yes
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
