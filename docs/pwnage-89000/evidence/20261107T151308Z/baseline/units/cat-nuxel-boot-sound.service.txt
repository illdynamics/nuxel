# /etc/systemd/system/nuxel-boot-sound.service
[Unit]
Description=Nuxel startup cue after confirmed boot-animation to launcher handoff
After=nuxel-wayland.service nuxel-wl-fb.service sound.target nuxel-button-manager.service
Wants=nuxel-wayland.service nuxel-wl-fb.service
ConditionPathExists=/usr/share/nuxel/assets/nuxel-pow.mp3
# Cheap early gate on the *live* route state.  nuxel-audio-route.service writes
# this file before it exits, so a boot where the route script never ran at all
# does not start the cue.  The state file is only a gate, never proof: the cue
# below re-runs the read-only --check immediately before playback.
ConditionPathExists=/run/nuxel-audio-route.state
# The sound-cue binary records audibility as unverified; no physical claim is
# made here.

[Service]
Type=oneshot
# --wait polls the SAME read-only checks (calliope MUB0, both DSP1s,
# every route control, 48 kHz sink, default sink, volume cap) for up to
# 200 s.  A one-shot --check here failed at 09:35:45Z, 0.1 s after the
# ABOX reload, before the CS35L41 DSP1 download had finished.
# Bounded: a cold ABOX DSP download must never hold the boot transaction
# open forever. The cue is cosmetic; the ABOX timer can finish later.
ExecStartPre=-/usr/lib/nuxel/nuxel-audio-route.sh --check
ExecStart=/usr/libexec/nuxel/nuxel-sound-cue boot
# The cue process itself repeats the read-only `nuxel-audio-route.sh --check`
# readiness gate immediately before playback; a stale state file is
# deliberately NOT sufficient.
# Playback is diagnostic until the physical route is verified. It must never
# hold the boot target hostage; nuxel-sound-cue itself bounds a player attempt
# to 11 seconds and reaps its process group.
# The cue waits (bounded, 180 s inside the program) for the live route check
# first.  A cold boot needs the ABOX PM reload plus both CS35L41 DSP1 firms to
# boot; the old 45 s service timeout killed the cue before that could finish,
# which is exactly what failed on 2026-10-30 09:35:45Z.  pw-play is separately
# bounded inside nuxel-sound-cue.
TimeoutStartSec=240
TimeoutStopSec=10
StandardOutput=journal
StandardError=journal
# Deliberately NOT Restart=always: play once, not on every bridge crash.

[Install]
WantedBy=nuxel-wayland.service

# /usr/lib/systemd/system/service.d/10-timeout-abort.conf
# This file is part of the systemd package.
# See https://fedoraproject.org/wiki/Changes/Shorter_Shutdown_Timer.
#
# To facilitate debugging when a service fails to stop cleanly,
# TimeoutStopFailureMode=abort is set to "crash" services that fail to stop in
# the time allotted. This will cause the service to be terminated with SIGABRT
# and a coredump to be generated.
#
# To undo this configuration change, create a mask file:
#   sudo mkdir -p /etc/systemd/system/service.d
#   sudo ln -sv /dev/null /etc/systemd/system/service.d/10-timeout-abort.conf

[Service]
TimeoutStopFailureMode=abort

# /etc/systemd/system/nuxel-boot-sound.service.d/20-fast-stop.conf
[Service]
TimeoutStopSec=5
KillMode=control-group
SendSIGKILL=yes

# /etc/systemd/system/nuxel-boot-sound.service.d/20-warmup-keepalive.conf
[Service]
# nuxel-audio-warmup holds an inaudible stream briefly so the 366 ms cue is
# not swallowed by a cold speaker DAPM path.  That helper is intentionally a
# short-lived background process; do not let oneshot teardown wait/kill the
# whole cgroup and mark a successful cue as failed.
KillMode=process
TimeoutStopSec=20

# /etc/systemd/system/nuxel-boot-sound.service.d/30-defer-abox.conf
# PWNAGE-51000: the boot cue is cosmetic.  It must not pull the 20-50 s ABOX
# firmware reload or the audio-route oneshot into the initial boot transaction.
# The ABOX timer runs after Home and starts the route when the DSP is ready.
[Unit]
After=nuxel-wayland.service nuxel-wl-fb.service sound.target nuxel-button-manager.service
Wants=nuxel-wayland.service nuxel-wl-fb.service

[Service]
# Fast, non-fatal readiness check; the cue itself rechecks before playback.
ExecStartPre=-/usr/lib/nuxel/nuxel-audio-route.sh --check

# /etc/systemd/system/nuxel-boot-sound.service.d/40-after-pipewire.conf
# The cue used to select its player at import time, before PipeWire had created
# /run/nuxel-wayland/pipewire-0, and then fell back to direct ALSA (EINVAL).
# Start it only after the standalone audio session, and treat this cosmetic cue
# as non-fatal so a cold ABOX/DSP route cannot leave a failed unit at boot.
[Unit]
After=nuxel-audio-session.service
Wants=nuxel-audio-session.service

[Service]
ExecStartPre=/usr/bin/sh -c 'for i in $(seq 1 50); do [ -S /run/nuxel-wayland/pipewire-0 ] && exit 0; sleep 0.2; done; exit 0'
ExecStart=
ExecStart=-/usr/libexec/nuxel/nuxel-sound-cue boot

# /etc/systemd/system/nuxel-boot-sound.service.d/99-p80000-fast-stop.conf
[Service]
# The boot cue is cosmetic and may still be waiting on the ABOX/DSP route when
# the owner reboots.  It must never hold the shutdown transaction for 20 s.
TimeoutStopSec=5s
# Keep the oneshot lifecycle attached to the cue process only.  20-warmup-
# keepalive.conf intentionally leaves a short-lived nuxel-audio-warmup helper
# in the cgroup; control-group teardown waits for it and previously marked a
# successful cue as failed after this 5 s timeout.
KillMode=process
SendSIGKILL=yes
