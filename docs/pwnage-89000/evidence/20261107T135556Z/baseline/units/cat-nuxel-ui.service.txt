# /etc/systemd/system/nuxel-ui.service
# Nuxel phone UI — the fbdev-native interactive interface for the S10+ (beyond2lte).
#
# A28: this process now starts on the NUXEL HOME LAUNCHER page.  The launcher's
# app registry, ordering, icons, colours and geometry are compiled in from
# pwa/nuxel-home/js/apps.js + css/style.css (tools/s10plus/a28/gen-launcher.py);
# the A23..A27 diagnostic pages are retained and reachable by tapping the
# launcher title or with `page=<p>` on /run/nuxel-ui.cmd.
#
# This kernel (Samsung Linux 4.14.253) has `# CONFIG_DRM is not set`, so there is
# no /dev/dri and no wlroots/Phoc/Phosh session.  /dev/fb0 through the DECON
# driver is the hardware-proven display path, and nuxel-ui is a real userspace
# process that owns it.
#
# ORDERING IS DELIBERATELY EMPTY (A24, measured on A23): the Fedora userspace of
# this image stalls before local-fs.target / multi-user.target -- every job sits in
# state "waiting" and nothing runs.  A unit carrying `After=systemd-udev-settle
# .service local-fs.target` therefore NEVER starts, and `ConditionPathExists=/dev/fb0`
# was additionally evaluated as failed even though /dev/fb0 exists (crw------- 29,0).
# The UI instead starts with no ordering, opens /dev/fb0 itself and retries via
# Restart, and nuxel-postboot.sh starts it again if it is ever missing.
[Unit]
Description=Nuxel Home launcher (fbdev-native platform adapter; the UTM PWA launcher on the S10+ panel)
Documentation=file:/usr/share/doc/nuxel/UTM-LAUNCHER-PORT.md
Conflicts=nuxel-firstlight.service

[Service]
Type=simple
ExecStart=/usr/bin/nuxel-ui
# Restart=always also covers a clean exit (a failed frame commit path exits 0 in
# --once mode only).  An explicit `systemctl stop` still stops it.
Restart=always
RestartSec=2
TimeoutStartSec=40
Nice=-5
OOMScoreAdjust=-500
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
WantedBy=sysinit.target

# /etc/systemd/system/nuxel-ui.service.d/10-launcher-primary.conf
# Nuxel S10+ — the A27 fbdev UI unit is NOT the panel owner on this image.
#
# WHY (A29b, measured live): /init re-enables nuxel-ui.service into BOTH
# sysinit.target.wants and multi-user.target.wants on every boot, and the
# initramfs post-handoff supervisor (tools/s10plus/a24/nuxel-postboot.sh) does
# `systemctl start nuxel-ui` every 30 s whenever it is not active.  With the
# UTM-derived launcher (nuxel-wayland + nuxel-wl-fb) enabled as the default
# screen, those two paths would keep starting a second /dev/fb0 owner behind the
# launcher's back — measured at 00:03:16 on the handset: `systemctl stop
# nuxel-ui` at 00:03:07, `Started nuxel-ui.service` at 00:03:16.
#
# `systemctl mask nuxel-ui` cannot be used: the unit file lives in
# /etc/systemd/system and systemd refuses to overwrite it.  A condition is the
# durable, non-destructive lever: the unit is *skipped* while the launcher bridge
# is present, so `systemctl start nuxel-ui` becomes a no-op that leaves the panel
# to the launcher.
#
# The A27 fb UI is NOT lost: nuxel-fallback.service still runs /usr/bin/nuxel-ui
# directly whenever the launcher fails to become ready, so exactly one
# framebuffer owner exists in either case.
[Unit]
ConditionPathExists=!/usr/lib/nuxel/nuxel-wl-fb

# /usr/lib/systemd/system/service.d/10-timeout-abort.conf
# This file is part of the systemd package.
# See https://fedoraproject.org/wiki/Changes/Shorter_Shutdown_Timer.
#
# To facilitate debugging when a service fails to stop cleanly,
# TimeoutStopFailureMode=abort is set to "crash" services that fail to stop in
# the time allotted. This will cause the service to be terminated with SIGABRT

[Service]
