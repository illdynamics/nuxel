    # /etc/systemd/system/nuxel-bridge.service
    [Unit]
    Description=Nuxel Bridge - PWA to Native Phone Service Mediator
    Documentation=https://github.com/nuxel/nuxel/docs/architecture.md
    # DEPENDENCY DIRECTION (measured A33): the bridge CALLS the notification hub
    # (SendNotification -> org.nuxel.Notifications on the system bus), so the hub must
    # be up first. The previous revision had `Before=nuxel-notifications.service`
    # here while nuxel-notifications.service had `Before=nuxel-bridge.service`,
    # which is a real ordering CYCLE (tools/s10plus/a33/verify-service-graph.py
    # failed on it). The hub is the producer, the bridge is the consumer:
    #   nuxel-notifications.service -> nuxel-bridge.service
    Wants=dbus.service ModemManager.service nuxel-portal.service nuxel-notifications.service
    After=dbus.service ModemManager.service nuxel-portal.service nuxel-notifications.service network-online.target
    Before=phosh.service
    # Restart-burst bound ([Unit], not [Service]).
    StartLimitIntervalSec=120
    StartLimitBurst=10
    [Service]
    Type=simple
    ExecStart=/usr/bin/nuxel-bridge
    # The bridge refuses to answer a PWA whose bridge_version does not match the
    # pinned value in pwa/nuxel-common/bridge-version.json; publish it so the pin is
    # visible in `systemctl show -p Environment`.
    #   NUXEL_BRIDGE_PROTOCOL_VERSION = the pinned PWA<->host contract ("1")
    #   NUXEL_BRIDGE_VERSION          = the host implementation version
    Environment=NUXEL_BRIDGE_PROTOCOL_VERSION=1
    Environment=NUXEL_BRIDGE_VERSION=0.6.1-beta
    Environment=NUXEL_PATHS_CONF=/etc/nuxel/paths.conf
    Restart=on-failure
    RestartSec=5
    RestartMaxDelaySec=30
    # Security hardening
    User=nuxel-bridge
    Group=nuxel-bridge
    # The portal owns the shared capability store as the graphical `nuxel` user.
    # The bridge also persists the grants it brokers, so grant it only group access
    # to that one directory rather than running the bridge as root.
    SupplementaryGroups=nuxel
    ProtectSystem=full
    ProtectHome=true
    PrivateTmp=true
    NoNewPrivileges=true
    CapabilityBoundingSet=
    AmbientCapabilities=
    SecureBits=keep-caps
    MemoryDenyWriteExecute=true
    RestrictRealtime=true
    RestrictNamespaces=true
    SystemCallArchitectures=native
    SystemCallFilter=@system-service @dbus
    SystemCallErrorNumber=EPERM
    LockPersonality=true
    PrivateDevices=true
    ProtectKernelTunables=true
    ProtectKernelModules=true
    ProtectKernelLogs=true
    ProtectControlGroups=true
    ProtectProc=invisible
    ProcSubset=pid
    RemoveIPC=true
    DevicePolicy=closed
    KeyringMode=private
    RuntimeDirectory=nuxel-bridge
    StateDirectory=nuxel-bridge
    CacheDirectory=nuxel-bridge
    # Permission grants and the HTTP-bridge token live in the shared Nuxel data dir
    # (configs/nuxel/paths.conf: NUXEL_DATA_DIR=/var/lib/nuxel).
    ReadWritePaths=/var/lib/nuxel /var/log/nuxel
    # D-Bus access
    # Sockets: D-Bus uses system bus socket, no extra sockets needed
    [Install]
    WantedBy=multi-user.target

    # /etc/systemd/system/nuxel-http-bridge.service
    [Unit]
    Description=Nuxel HTTP Bridge API
    After=dbus.service nuxel-bridge.service
    Wants=dbus.service nuxel-bridge.service
    [Service]
    Type=simple
    ExecStart=/usr/bin/nuxel-http-bridge
    Restart=on-failure
    RestartSec=3
    # S10+ vendor kernel (4.14.253) ncm module denies AF_INET/AF_INET6 socket
    # creation to non-root UIDs; this HTTP API binds 127.0.0.1:8080, so it must
    # run as root (still sandboxed below). Documented kernel limitation, not a
    # relaxed policy choice.
    User=root
    Group=root
    NoNewPrivileges=yes
    PrivateTmp=yes
    ProtectSystem=strict
    ProtectHome=yes
    PrivateDevices=yes
    RestrictAddressFamilies=AF_INET AF_UNIX
    ReadOnlyPaths=/
    ReadWritePaths=/var/lib/nuxel /var/log/nuxel
    LogsDirectory=nuxel
    Environment=NUXEL_HTTP_BIND=127.0.0.1
    Environment=NUXEL_HTTP_PORT=8080
    Environment=NUXEL_HTTP_MODE=dev
    Environment=NUXEL_PATHS_CONF=/etc/nuxel/paths.conf
    # The HTTP bridge reads the SAME canonical app registry the PWA and the native
    # shell use, so /api/v1/apps/registry cannot drift from the launcher.
    Environment=NUXEL_APP_REGISTRY=/usr/share/nuxel/pwa/nuxel-common/app-registry.json
    Environment=NUXEL_BRIDGE_PROTOCOL_VERSION=1
    Environment=NUXEL_BRIDGE_VERSION=0.4.0-alpha
    Environment=NUXEL_API_ALLOW_ORIGIN=http://127.0.0.1:8000,http://localhost:8000
    [Install]
    WantedBy=multi-user.target
