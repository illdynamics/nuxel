# /etc/systemd/system/nuxel-wayland.service
[Unit]
Description=Nuxel phone Wayland session (phoc headless + Nuxel Home launcher)
Documentation=file:/usr/share/doc/nuxel/UTM-LAUNCHER-PORT.md
# Bounded restarts: never spin into a process storm.
StartLimitIntervalSec=60
StartLimitBurst=5
After=nuxel-touch-arm.service seatd.service nuxel-web.service
Wants=nuxel-touch-arm.service seatd.service nuxel-web.service
Before=nuxel-wl-fb.service nuxel-fallback.service

[Service]
Type=simple
# phoc with wlroots 0.19.  The wrapper tries Mali-G76 GLES2 through Mesa
# panfrost on ARM kbase first, and falls back to pixman if that path fails.
Environment=WLR_BACKENDS=headless,libinput
# The handset has NO VT (/dev/tty0, /sys/class/vtconsole, /proc/consoles all
# absent), so a VT-bound seat can never activate; libseat's noop backend
# satisfies wlroots' session requirement and device access is granted by the
# preloaded libinput path shim.  See tools/s10plus/a31/nuxel-libinput-path.c.
Environment=LIBSEAT_BACKEND=noop
# If the panel node is absent (charger mode: sec_ts never binds) wlroots would
# fail the whole backend and phoc would restart-loop.  The shim records what it
# found in /run/nuxel-libinput-path.state, so "zero devices" stays visible.
Environment=WLR_LIBINPUT_NO_DEVICES=1
# The udev daemon is masked on this image (it cannot register workers on 4.14),
# so libinput's udev backend discovers zero devices; the shim redirects ONLY
# libinput's discovery step to its path backend and adds the real
# /dev/input/eventN node that carries INPUT_PROP_DIRECT.  Real evdev, real
# libinput parsing, real wl_touch - no synthesised events.
Environment=LD_PRELOAD=/usr/lib/nuxel/libnuxel-libinput-path.so
Environment=WLR_HEADLESS_OUTPUTS=1
Environment=XDG_RUNTIME_DIR=/run/nuxel-wayland
Environment=XDG_SESSION_TYPE=wayland
ExecStartPre=/bin/mkdir -p /run/nuxel-wayland
ExecStartPre=/bin/chmod 700 /run/nuxel-wayland
ExecStart=/usr/lib/nuxel/nuxel-wayland-gpu-wrapper.sh
Restart=on-failure
RestartSec=3

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/nuxel-wayland.service.d/30-fast-stop.conf
# PWNAGE-37000 BUTTERSCOTCH boot-latency fix.
#
# The ABOX firmware-reload service stops nuxel-wayland.service before cycling
# the audio DSP, then restarts it. On this 4.14 vendor kernel the default
# 90 s stop timeout let phoc/WebKit/PipeWire sit in final-sigterm for ~46 s
# before systemd aborted the stop. That single stop is the largest measured
# component of the S10+ cold-boot "spinning penguin" delay.
#
# The session is restarted immediately afterwards by the ABOX wrapper, so a
# short, deterministic stop is the correct behaviour: TERM the whole cgroup,
# then SIGKILL what remains after 8 s. Normal shutdown also benefits because
# the compositor no longer hangs the reboot for up to 90 s.
[Service]
TimeoutStopSec=8
SendSIGKILL=yes
KillMode=control-group

# /etc/systemd/system/nuxel-wayland.service.d/40-bounded-stop.conf
# PWNAGE-51000: the compositor is the only unit that consumed its full stop
# timeout during a normal reboot.  The unit's MAINPID is the wrapper shell;
# phoc is its child, so signal both the main process and its children, then
# use a short unit-specific SIGKILL fallback.  Global timeouts are untouched.
[Service]
TimeoutStopSec=3
ExecStop=/bin/sh -c 'kill -TERM "$MAINPID" 2>/dev/null || true; pkill -TERM -P "$MAINPID" 2>/dev/null || true; n=0; while kill -0 "$MAINPID" 2>/dev/null && [ "$n" -lt 12 ]; do n=$((n+1)); sleep 0.1; done; pkill -KILL -P "$MAINPID" 2>/dev/null || true; kill -KILL "$MAINPID" 2>/dev/null || true'
KillMode=control-group
SendSIGKILL=yes

# /etc/systemd/system/nuxel-wayland.service.d/50-kbase-egl.conf
# PWNAGE-53000: load the kbase Mesa EGL build by absolute path.
# Without this, glvnd follows /usr/lib64/libEGL_mesa.so.0 and a Mesa package
# update can silently point it back at the stock (llvmpipe-capable) library.
[Service]
Environment=__EGL_VENDOR_LIBRARY_FILENAMES=/etc/nuxel/egl-kbase.json
Environment=LIBGL_DRIVERS_PATH=/usr/lib64/dri
